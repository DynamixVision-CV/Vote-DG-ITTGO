// ===== CONFIG =====
const ADMIN_CODE = 'ITTGO_VOTE-2026'; // <-- change ce code admin !
const NB_CODES = 500;

const sh = n => SpreadsheetApp.getActive().getSheetByName(n);
const out = o => ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);

// À exécuter UNE SEULE FOIS : crée les feuilles et les 500 codes
function setup() {
  const ss = SpreadsheetApp.getActive();
  ['Codes', 'Candidats', 'Votes'].forEach(n => { if (!ss.getSheetByName(n)) ss.insertSheet(n); });
  sh('Candidats').getRange(1, 1, 1, 5).setValues([['id', 'nom', 'classe', 'description', 'photo']]);
  sh('Votes').getRange(1, 1).setValue('candidatId');
  sh('Codes').clearContents();
  sh('Codes').getRange(1, 1, 1, 2).setValues([['code', 'utilise']]);
  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  const set = {};
  while (Object.keys(set).length < NB_CODES) {
    let c = '';
    for (let i = 0; i < 8; i++) c += chars[Math.floor(Math.random() * chars.length)];
    set[c] = 1;
  }
  const rows = Object.keys(set).map(c => [c, 'NON']);
  sh('Codes').getRange(2, 1, rows.length, 2).setValues(rows);
  // Suivi visuel : vert = non utilisé, rouge = utilisé
  const rng = sh('Codes').getRange('B2:B' + (NB_CODES + 1));
  sh('Codes').setConditionalFormatRules([
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('NON').setBackground('#c8f0d2').setRanges([rng]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('OUI').setBackground('#f6c9c9').setRanges([rng]).build()
  ]);
  sh('Codes').getRange('D1:E3').setFormulas([
    ['Total', '=COUNTA(A2:A)'],
    ['Utilisés', '=COUNTIF(B2:B,"OUI")'],
    ['Non utilisés', '=COUNTIF(B2:B,"NON")']
  ]);
  sh('Codes').setColumnWidth(1, 140);
}

function getCandidats() {
  const v = sh('Candidats').getDataRange().getValues();
  v.shift();
  return v.filter(r => r[0] !== '').map(r => ({
    id: String(r[0]), nom: r[1], classe: r[2], description: r[3], photo: r[4]
  }));
}

function doGet() { return out({ ok: true, candidats: getCandidats() }); }

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const d = JSON.parse(e.postData.contents);
    if (d.action === 'vote') return out(vote(d));
    if (d.adminCode !== ADMIN_CODE) return out({ ok: false, error: 'Code admin invalide' });
    if (d.action === 'login') return out({ ok: true });
    if (d.action === 'results') return out(results());
    if (d.action === 'add') {
      const id = Utilities.getUuid().slice(0, 8);
      sh('Candidats').appendRow([id, d.nom, d.classe, d.description, d.photo || '']);
      return out({ ok: true });
    }
    if (d.action === 'delete') {
      const v = sh('Candidats').getDataRange().getValues();
      for (let i = 1; i < v.length; i++) {
        if (String(v[i][0]) === String(d.id)) { sh('Candidats').deleteRow(i + 1); break; }
      }
      return out({ ok: true });
    }
    return out({ ok: false, error: 'Action inconnue' });
  } catch (err) {
    return out({ ok: false, error: String(err) });
  } finally {
    lock.releaseLock();
  }
}

function vote(d) {
  const code = String(d.code || '').trim().toUpperCase();
  const codes = sh('Codes').getDataRange().getValues();
  let row = -1;
  for (let i = 1; i < codes.length; i++) if (String(codes[i][0]) === code) { row = i; break; }
  if (row < 0) return { ok: false, error: 'Code invalide' };
  if (codes[row][1] === 'OUI') return { ok: false, error: 'Ce code a déjà été utilisé' };
  if (!getCandidats().some(c => c.id === String(d.candidatId))) return { ok: false, error: 'Candidat introuvable' };
  sh('Votes').appendRow([String(d.candidatId)]); // anonyme : aucun lien avec le code
  sh('Codes').getRange(row + 1, 2).setValue('OUI');
  return { ok: true };
}

function results() {
  const votes = sh('Votes').getDataRange().getValues().slice(1).map(r => String(r[0]));
  const counts = {};
  votes.forEach(v => counts[v] = (counts[v] || 0) + 1);
  const used = sh('Codes').getDataRange().getValues().slice(1).filter(r => r[1] === 'OUI').length;
  return {
    ok: true, used, total: NB_CODES,
    candidats: getCandidats().map(c => Object.assign(c, { votes: counts[c.id] || 0 }))
  };
}
